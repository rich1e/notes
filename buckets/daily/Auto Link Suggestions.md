## 🛠️ 批量添加链接脚本

以下 Python 脚本可以自动为文档添加相关文献链接：

```python
#!/usr/bin/env python3
import os
import re
from collections import defaultdict

DAILY_DIR = '/Users/rich1e/workspace/code/notes/buckets/daily'

def extract_tags_and_content(filepath):
    """提取文件中的标签和内容"""
    with open(filepath, 'r', encoding='utf-8') as f:
        content = f.read()

    # 匹配 #tag 格式的标签
    tags = set(re.findall(r'#[a-zA-Z0-9_-]+', content))
    tags = {tag[1:] for tag in tags if len(tag) > 2 and not tag[1:].isdigit()}

    return tags, content

def build_tag_index():
    """构建标签索引"""
    tag_index = defaultdict(list)

    for filename in sorted(os.listdir(DAILY_DIR)):
        if not filename.endswith('.md') or filename.startswith('.') or 'Tag Links' in filename or 'Auto Link' in filename:
            continue

        filepath = os.path.join(DAILY_DIR, filename)
        tags, _ = extract_tags_and_content(filepath)

        for tag in tags:
            tag_index[tag].append(filename.replace('.md', ''))

    return tag_index

def generate_related_section(filename, tag_index, current_tags):
    """生成相关阅读部分"""
    related_files = set()

    for tag in current_tags:
        files_with_tag = tag_index.get(tag, [])
        for other_file in files_with_tag:
            if other_file != filename.replace('.md', ''):
                related_files.add(other_file)

    if not related_files:
        return ""

    # 限制最多显示5个相关文档
    top_related = sorted(related_files)[:5]

    section = "\n\n## 🔗 相关阅读\n"
    for related in top_related:
        section += f"- [[{related}]]\n"

    return section

def add_links_to_file(filepath, tag_index):
    """为单个文件添加链接"""
    filename = os.path.basename(filepath)
    tags, content = extract_tags_and_content(filepath)

    # 检查是否已有"相关阅读"部分
    if '## 🔗 相关阅读' in content or '## 相关阅读' in content:
        print(f"  ⏭️  跳过 {filename} (已有相关链接)")
        return False

    # 生成相关链接
    related_section = generate_related_section(filename, tag_index, tags)

    if not related_section:
        print(f"  ❌ 无相关文档 {filename}")
        return False

    # 在内容末尾添加（最后一个 --- 分隔线之后，或文件末尾）
    if '\n---\n' in content:
        parts = content.rsplit('\n---\n', 1)
        new_content = parts[0] + '\n---\n' + related_section + parts[1] if len(parts) > 1 else content + related_section
    else:
        new_content = content.rstrip() + related_section

    # 写入文件
    with open(filepath, 'w', encoding='utf-8') as f:
        f.write(new_content)

    print(f"  ✅ 已更新 {filename}")
    return True

def main():
    print("🔍 正在扫描 daily 目录...")
    tag_index = build_tag_index()

    print(f"📊 发现 {len(tag_index)} 种标签\n")

    # 统计有多少文件需要更新
    updated_count = 0
    total_files = 0

    for filename in sorted(os.listdir(DAILY_DIR)):
        if not filename.endswith('.md') or filename.startswith('.') or 'Tag Links' in filename or 'Auto Link' in filename:
            continue

        total_files += 1
        filepath = os.path.join(DAILY_DIR, filename)
        if add_links_to_file(filepath, tag_index):
            updated_count += 1

    print(f"\n{'='*60}")
    print(f"✅ 完成！更新了 {updated_count}/{total_files} 个文件")
    print(f"💡 建议使用 Git 查看变更并分批提交")

if __name__ == '__main__':
    main()
```

````markdown
```dataviewjs
// 获取当前文件的标签
const currentFile = dv.current().file;
const currentTags = currentFile.tags || [];

// 查找具有相同标签的其他文件
const relatedFiles = dv.pages('"buckets/daily"')
    .where(p => p.file.name !== currentFile.name)
    .where(p => {
        const fileTags = p.file.tags || [];
        return currentTags.some(tag => fileTags.includes(tag));
    })
    .slice(0, 5);

if (relatedFiles.length > 0) {
    dv.header(2, "🔗 相关阅读");
    dv.list(relatedFiles.map(p => p.file.link));
}
```
````
